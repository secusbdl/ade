cmake_minimum_required(VERSION 3.25)
project(AdeGitUsageStage LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Test)
qt_standard_project_setup()
add_compile_options(-Wall -Wextra -Wpedantic)
add_link_options(-static -static-libgcc -static-libstdc++)
add_library(ade_git_usage STATIC src/git/WorktreeUsage.h src/git/WorktreeUsage.cpp src/git/WorktreeUsageJournal.h src/git/WorktreeUsageJournal.cpp src/git/WorktreeUsageIdentity.h src/git/WorktreeUsageIdentity.cpp)
target_include_directories(ade_git_usage PUBLIC src)
target_link_libraries(ade_git_usage PUBLIC Qt6::Core)
qt_add_executable(test_git_usage tests/test_git_usage.cpp)
target_link_libraries(test_git_usage PRIVATE ade_git_usage Qt6::Test)
