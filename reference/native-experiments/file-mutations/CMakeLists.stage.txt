cmake_minimum_required(VERSION 3.25)
project(FileMutationsStage LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_LINK_SEARCH_END_STATIC TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Sql Test)
add_compile_options(-Wall -Wextra -Wpedantic)
add_link_options(-static -static-libgcc -static-libstdc++)
add_library(mutation_stage STATIC src/FileMutationStore.cpp src/FileMutationStore.h)
target_include_directories(mutation_stage PUBLIC src)
target_link_libraries(mutation_stage PUBLIC Qt6::Core Qt6::Sql)
add_executable(test_file_mutations tests/test_file_mutations.cpp)
target_link_libraries(test_file_mutations PRIVATE mutation_stage Qt6::Test)
