cmake_minimum_required(VERSION 3.25)
project(AdeAppsProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_LINK_SEARCH_END_STATIC TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network Test)
add_executable(ade-app-resource-tests ../../tests/test_app_resources.cpp ../../src/apps/AppResourceStore.cpp ../../src/apps/AppResourceStore.h)
target_include_directories(ade-app-resource-tests PRIVATE ../../src)
target_link_libraries(ade-app-resource-tests PRIVATE Qt6::Core Qt6::Test)
target_link_options(ade-app-resource-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_executable(ade-mini-app-tests ../../tests/test_mini_apps.cpp ../../src/apps/MiniAppHost.cpp ../../src/apps/MiniAppHost.h ../../src/apps/MiniAppPage.cpp ../../src/apps/MiniAppPage.h ../../src/apps/AppResourceStore.cpp ../../src/browser/browserwidget.cpp ../../src/browser/browserwidget.h)
target_include_directories(ade-mini-app-tests PRIVATE ../../src)
target_include_directories(ade-mini-app-tests SYSTEM PRIVATE ../../third_party/webview2/include)
target_link_libraries(ade-mini-app-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test ole32)
target_link_options(ade-mini-app-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_custom_command(TARGET ade-mini-app-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/webview2/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:ade-mini-app-tests>")
add_executable(ade-sandbox-app-tests ../../tests/test_sandbox_apps.cpp ../../src/apps/MiniAppHost.cpp ../../src/apps/MiniAppHost.h ../../src/apps/MiniAppPage.cpp ../../src/apps/MiniAppPage.h ../../src/apps/AppResourceStore.cpp ../../src/browser/browserwidget.cpp ../../src/browser/browserwidget.h ../../src/browser/BrowserToolBridge.cpp ../../src/browser/BrowserToolBridge.h ../../src/sandbox/SandboxHost.cpp ../../src/sandbox/SandboxHost.h)
target_include_directories(ade-sandbox-app-tests PRIVATE ../../src)
target_include_directories(ade-sandbox-app-tests SYSTEM PRIVATE ../../third_party/webview2/include)
target_link_libraries(ade-sandbox-app-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test ole32)
target_link_options(ade-sandbox-app-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_custom_command(TARGET ade-sandbox-app-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/webview2/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:ade-sandbox-app-tests>" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../sandbox-probe-build/ADE-sandbox-worker.exe" "$<TARGET_FILE_DIR:ade-sandbox-app-tests>")
