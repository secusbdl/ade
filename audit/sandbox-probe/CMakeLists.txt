cmake_minimum_required(VERSION 3.25)
project(AdeSandboxProbe LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_LINK_SEARCH_END_STATIC TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test Concurrent ZlibPrivate)
set(QJS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/quickjs")
add_library(ade_quickjs STATIC "${QJS_ROOT}/dtoa.c" "${QJS_ROOT}/libregexp.c" "${QJS_ROOT}/libunicode.c" "${QJS_ROOT}/quickjs.c")
target_compile_definitions(ade_quickjs PRIVATE QUICKJS_NG_BUILD _GNU_SOURCE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(ade_quickjs PUBLIC "${QJS_ROOT}")
target_compile_options(ade_quickjs PRIVATE -funsigned-char)
add_executable(ade-quickjs-smoke quickjs-smoke.cpp)
target_link_libraries(ade-quickjs-smoke PRIVATE ade_quickjs)
target_link_options(ade-quickjs-smoke PRIVATE -static -static-libgcc -static-libstdc++)
add_executable(ADE-sandbox-worker ../../src/sandbox/worker_main.cpp)
target_link_libraries(ADE-sandbox-worker PRIVATE Qt6::Core ade_quickjs)
target_link_options(ADE-sandbox-worker PRIVATE -static -static-libgcc -static-libstdc++)
add_library(ade_sandbox STATIC ../../src/sandbox/SandboxHost.cpp ../../src/sandbox/SandboxHost.h)
target_include_directories(ade_sandbox PUBLIC ../../src)
target_link_libraries(ade_sandbox PUBLIC Qt6::Core)
add_executable(ade-sandbox-tests ../../tests/test_sandbox.cpp ../../src/core/ToolExecutor.cpp ../../src/core/ToolExecutor.h
    ../../src/core/ScopedDirectory.cpp ../../src/core/ScopedDirectory.h
    ../../src/attachments/FileTransformer.cpp ../../src/attachments/archive/ArchiveTransformer.cpp)
target_link_libraries(ade-sandbox-tests PRIVATE ade_sandbox Qt6::Test Qt6::Concurrent Qt6::ZlibPrivate)
target_link_options(ade-sandbox-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_dependencies(ade-sandbox-tests ADE-sandbox-worker)
add_executable(ade-sandbox-network-tests ../../tests/test_sandbox_network.cpp ../../src/network/NetworkBroker.cpp ../../src/network/NetworkBroker.h)
target_link_libraries(ade-sandbox-network-tests PRIVATE ade_sandbox Qt6::Network Qt6::Test)
target_link_options(ade-sandbox-network-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_dependencies(ade-sandbox-network-tests ADE-sandbox-worker)
